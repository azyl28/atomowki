# VENOL ATOMOWKI LODZ - strona internetowa

Kompletna, statyczna strona klubu kobiecej druzyny rugby **Venol Atomowki Lodz**.

## Uruchomienie

1. Skopiuj caly folder `Atomowki-Lodz/` na dysk.
2. Otworz plik `index.html` w dowolnej nowoczesnej przegladarce (Chrome, Firefox, Edge, Safari).
3. Nie wymaga Node.js, serwera ani frameworkow.

> Uwaga: w niektorych przegladarkach pliki lokalne (`file://`) moga blokowac czesc funkcji.
> Jesli chcesz miec pelna pewnosc, uruchom prosty serwer statyczny, np.:
> `python -m http.server 8000` - i wejdz na `http://localhost:8000`.

## Struktura

- `index.html` - strona glowna
- `pages/` - podstrony: o klubie, druzyna, akademia, rozgrywki, sukcesy, aktualnosci, galeria, kontakt
- `css/` - `style.css`, `animations.css`, `responsive.css`
- `js/` - `data.js` (dane), `main.js` (logika wspolna), `players.js`, `gallery.js`
- `assets/` - miejsce na zdjecia, logotypy, ikony

## Edycja tresci

Wszystkie dane klubowe znajduja sie w pliku **`js/data.js`** w obiekcie `ATOMOWKI_DATA`:

- `players` - zawodniczki (kategorie: `senior`, `u16`, `u13`)
- `news` - aktualnosci
- `successes` - sukcesy
- `matches` - mecze i wyniki
- `league` - tabela rozgrywek
- `gallery` - zdjecia
- `academy` - treningi i FAQ

## Brakujace materialy (do uzupelnienia przez klub)

- Oryginalne zdjecia druzyny, zawodniczek, meczow, treningow
- Logotypy: Venol, BBRC Lodz, sponsorzy i partnerzy
- Imiona i nazwiska zawodniczek, numery, pozycje
- Terminy i miejsca treningow
- Terminy meczow, wyniki, tabele
- Potwierdzone dane kontaktowe (adres, telefon, e-mail, Facebook, strona klubu)
- Pelna polityka prywatnosci
- Konfiguracja wysylki formularza (backend / Formspree)
- Mapa lokalizacji (osadzenie Google Maps / OpenStreetMap)

## Formularz kontaktowy

Formularz waliduje pola, ale **nie wysyla wiadomosci** - nie ma backendu.
Aby wysylac wiadomosci, podlacz zewnetrzna usluge (np. Formspree) lub wlasny endpoint.

## Licencja

Kod przygotowany na potrzeby klubu Venol Atomowki Lodz.

/* =========================================================
   gallery.js - siatka galerii + lightbox.
   ========================================================= */
(function () {
  "use strict";
  const grid = document.getElementById("galleryGrid");
  if (!grid) return;

  const D = window.ATOMOWKI_DATA || { gallery: [] };
  const BASE = document.body.dataset.base || "";
  let currentCat = "all";
  let visibleItems = [];
  let currentIndex = 0;

  function render() {
    visibleItems = (D.gallery || []).filter(function(g){ return currentCat === "all" || g.category === currentCat; });
    grid.innerHTML = visibleItems.map(function(g, i){
      return '<button class="gallery-item" data-gallery-index="' + i + '" aria-label="' + g.caption + '">' +
        (g.src ? '<img src="' + BASE + g.src + '" alt="' + g.caption + '" loading="lazy">' : "MIEJSCE NA ZDJECIE") +
        '<span class="gallery-cat">' + g.category + '</span>' +
      '</button>';
    }).join("");
    grid.querySelectorAll("[data-gallery-index]").forEach(function(btn){
      btn.addEventListener("click", function(){ openLightbox(Number(btn.dataset.galleryIndex)); });
    });
  }

  const filters = document.getElementById("galleryFilters");
  if (filters) {
    filters.addEventListener("click", function(e){
      const btn = e.target.closest(".filter");
      if (!btn) return;
      filters.querySelectorAll(".filter").forEach(function(b){ b.classList.remove("active"); });
      btn.classList.add("active");
      currentCat = btn.dataset.cat;
      render();
    });
  }

  const lb = document.getElementById("lightbox");
  const lbImg = document.getElementById("lightboxImage");
  const lbCap = document.getElementById("lightboxCaption");
  if (!lb || !lbImg) return;

  function openLightbox(index) {
    currentIndex = index;
    updateLightbox();
    lb.classList.add("open");
    lb.setAttribute("aria-hidden", "false");
    document.body.style.overflow = "hidden";
  }
  function closeLightbox() {
    lb.classList.remove("open");
    lb.setAttribute("aria-hidden", "true");
    document.body.style.overflow = "";
  }
  function updateLightbox() {
    const item = visibleItems[currentIndex];
    if (!item) return;
    lbImg.innerHTML = item.src
      ? '<img src="' + BASE + item.src + '" alt="' + item.caption + '">'
      : '<span>MIEJSCE NA ZDJECIE</span>';
    lbCap.textContent = item.caption + ' (' + (currentIndex + 1) + '/' + visibleItems.length + ')';
  }
  function next() { currentIndex = (currentIndex + 1) % visibleItems.length; updateLightbox(); }
  function prev() { currentIndex = (currentIndex - 1 + visibleItems.length) % visibleItems.length; updateLightbox(); }

  lb.querySelector("[data-close-lightbox]").addEventListener("click", closeLightbox);
  lb.querySelector("[data-lightbox-next]").addEventListener("click", next);
  lb.querySelector("[data-lightbox-prev]").addEventListener("click", prev);
  document.addEventListener("keydown", function(e){
    if (!lb.classList.contains("open")) return;
    if (e.key === "Escape") closeLightbox();
    if (e.key === "ArrowRight") next();
    if (e.key === "ArrowLeft") prev();
  });

  render();
})();

  function initHeroSlider() {
    const imgs = document.querySelectorAll('.hero-media .slider-img');
    if (!imgs.length) return;

    const dotsContainer = document.querySelector('.slider-dots');
    const prevBtn = document.querySelector('.slider-arrow.prev');
    const nextBtn = document.querySelector('.slider-arrow.next');
    
    let current = 0;

    // Generujemy kropki
    if (dotsContainer) {
      dotsContainer.innerHTML = '';
      imgs.forEach(function(_, i) {
        const dot = document.createElement('span');
        dot.className = 'dot' + (i === 0 ? ' active' : '');
        dot.addEventListener('click', function() { goToSlide(i); });
        dotsContainer.appendChild(dot);
      });
    }

    function goToSlide(index) {
      if (index < 0) index = imgs.length - 1;
      if (index >= imgs.length) index = 0;
      imgs[current].classList.remove('active');
      current = index;
      imgs[current].classList.add('active');
      if (dotsContainer) {
        dotsContainer.querySelectorAll('.dot').forEach(function(d, i) {
          d.classList.toggle('active', i === current);
        });
      }
    }

    if (prevBtn) prevBtn.addEventListener('click', function() { goToSlide(current - 1); });
    if (nextBtn) nextBtn.addEventListener('click', function() { goToSlide(current + 1); });

    // Automatyczne przewijanie
    setInterval(function() {
      goToSlide(current + 1);
    }, 5000);
  }

  // Uruchamiamy slider
  initHeroSlider();
